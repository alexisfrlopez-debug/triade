lemma implication_secret_appris_reconstitution_illicite:
  all-traces
  "∀ p #i #k.
    ((PhraseVivante( p ) @ #k) ∧ (K( p ) @ #i)) ⇒
    ((∃ #j. (ReconstitutionIllicite( p ) @ #j) ∧ (#j < #i)) ∧
     (∃ x1 x2 x3 #j1 #j2 #j3.
       (((((PartTenue( x1 ) @ #j1) ∧ (PartTenue( x2 ) @ #j2)) ∧
          (PartTenue( x3 ) @ #j3)) ∧
         (#j1 < #i)) ∧
        (#j2 < #i)) ∧
       (#j3 < #i)))"
/*
guarded formula characterizing all counter-examples:
"∃ p #i #k.
  (PhraseVivante( p ) @ #k) ∧ (K( p ) @ #i)
 ∧
  ((∀ #j. (ReconstitutionIllicite( p ) @ #j) ⇒ ¬(#j < #i)) ∨
   (∀ x1 x2 x3 #j1 #j2 #j3.
     (PartTenue( x1 ) @ #j1) ∧
     (PartTenue( x2 ) @ #j2) ∧
     (PartTenue( x3 ) @ #j3)
    ⇒
     ((¬(#j1 < #i)) ∨ (¬(#j2 < #i)) ∨ (¬(#j3 < #i)))))"
*/
