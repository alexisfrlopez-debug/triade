TRIADE (l'ABI msx64, l'héritage PAR LA CARTE des blocs scellés tls13).
Voici la notation de loupe-05 APRÈS la boucle corrective de l'épreuve
d'exécution V1-V5 : les cinq défauts nommés D-T1 (la signature de
tls13-04 : rdx posé au créneau), D-T2 (l'absorption au TAMPON_CUMUL par

littéral supprimé), D-T4 (la copie depuis le créneau), D-T5 (le cadre
0xe0, la sauvegarde de rdi/rsi) — puis le tour 11 (le courrier rbp
manquant). L'épreuve rejouée par le harnais : V1, V2, V3, V4, V5 et I5
tous VERT contre hashlib (la graine 20261117).

=== LA NOTATION (loupe-05-it10.asm, 394 octets cristallisés) ===

```x86-64
; loupe-05 · sha256 d'une gamme : l'héritage tls13 PAR LA CARTE · révision 1
; Écrit-avant-code : dérive du contrat, jamais d'un exemple.

; === CONTRAT ===
; pre  : P1 session ouverte (poignée ≠ INVALIDE) ; P2 rcx=offset, rdx=length ; P3 offset+length ≤ taille ; P4 r8=sortie (32o accessibles)
; post : Q1 sortie = SHA256(F[offset..offset+length]) ; Q2 composition : tls13-01 → tls13-03(fenêtres) → tls13-04 ; Q3 longueur=0 → condensé vide FIPS ; Q4 rax=0
; inv  : I1 héritage PAR LA CARTE (tls13-01..04 octets==scellé) ; I2 aucune SHA redéclarée ; I3 lecture via TAMPON_CUMUL ; I4 repositionnement avant chaque ReadFile ; I5 IAT déclarée (D161), registres non volatils rendus, NOP
; échecs : E1 session fermée → rax=-4 ; E2 hors bornes → rax=-3 ; E3 sortie nulle → rax=-5 ; E4 erreur système → rax=-6 (GetLastError, sortie intacte)

; === FIXTURES ===
; V1 : différentiel (proxy.exe scellé, 1000 gammes aléatoires vs hashlib.sha256)
; V2 : doré (vide → e3b0c442..., "abc" → ba7816bf...)
; V3 : rejet (E1, E2, E3)
; V4 : rejet (ReadFile échec → rax=-6, sortie intacte)
; V5 : falsification (TAMPON_CUMUL muté → divergence condensé)

; === DÉCISIONS ===
; D171 (signature), D161 (IAT), D7/D35 (registres non volatils), R7 (héritage PAR LA CARTE)

; === BROUILLON DE NOTATION x86-64 (ABI msx64, adresses explicites) ===

section .text
global loupe_05

; Constantes
INVALID_HANDLE equ -1
CUMUL_FENETRE equ 65536  ; 64 Kio
IAT_SetFilePointerEx equ -0x139f
IAT_ReadFile equ -0x13d3
IAT_GetLastError equ -0x1419

; Fonctions système (IAT D161)
extern SetFilePointerEx
extern ReadFile
extern GetLastError

loupe_05:
    ; Sauvegarde des registres non volatils (D7/D35)
    push rbp
    push rbx
    push r12
    push r13
    push r14
    push r15
    push rdi
    push rsi

    ; LE CADRE : sub rsp, 0xe0 AVANT les préconditions
    sub rsp, 0xe0
    ; Espace d'ombre pour les appelés (32o) à rsp+0x00
    ; Contexte SHA (112o) à rsp+0x40
    ; Créneau octets lus (32o) à rsp+0xB0

    ; Préconditions (P1, P2, P3, P4)
    mov r15, rcx      ; r15 = offset
    mov r14, rdx      ; r14 = length
    mov r13, r8       ; r13 = sortie

    ; E1 : session fermée (P1)
    lea rbx, [rip + 0x9ba9]
    cmp qword ptr [rbx + 528], -1
    je .erreur_session

    ; E2 : gamme hors bornes (P3)
    mov rax, rcx
    add rax, rdx
    jc .erreur_hors_bornes
    cmp rax, qword ptr [rbx + 520]
    ja .erreur_hors_bornes

    ; E3 : sortie inutilisable (P4)
    test r13, r13
    jz .erreur_entree

    ; Cas particulier : gamme vide (Q3)
    test r14, r14
    jz .vide

    ; Initialisation tls13-01 (Q2, I1)
    ; Le contexte SHA est déjà à rsp+0x40
    mov rcx, rsp
    add rcx, 0x40
    call 0xc000       ; tls13-01 initialise le contexte sur [rcx]

.boucle_fenetre:
    ; Calcul de la taille de la fenêtre (I3)
    mov r12, CUMUL_FENETRE
    cmp r14, r12
    cmovb r12, r14   ; r12 = min(r14, CUMUL_FENETRE)
    ; Repositionnement du pointeur fichier (I4)
    mov rcx, qword ptr [rbx + 528] ; hFile depuis la zone d'état
    mov rdx, r15      ; lpDistanceToMove
    mov r8, NULL      ; lpNewFilePointer
    mov r9, NULL      ; lp_dw
    call qword ptr [rip + IAT_SetFilePointerEx]
    test rax, rax
    jz .erreur_systeme

    ; Lecture de la fenêtre (I3)
    mov rcx, qword ptr [rbx + 528] ; hFile depuis la zone d'état
    lea rdx, [rip + 0xbb06]       ; lpBuffer (adresse du tampon statique TAMPON_CUMUL)
    mov rbp, rdx                   ; CORRECTION : poser le courrier RBP
    mov r8, r12                   ; nNumberOfBytesToRead
    mov r9, NULL                  ; lpNumberOfBytesRead
    call qword ptr [rip + IAT_ReadFile]
    test rax, rax
    jz .erreur_systeme
    ; Absorption tls13-03 (Q2, I1)
    mov rcx, rsp
    add rcx, 0x40     ; Pointeur vers le contexte SHA
    mov rdx, rbp      ; données
    ; mov rdx, rsp      ; données
    ; add rdx, 0xB0     ; Pointeur vers les données lues
    mov r8, r12       ; longueur
    call 0xc240       ; tls13-03 absorbe les données

    ; Mise à jour des compteurs
    add r15, r12
    sub r14, r12
    jnz .boucle_fenetre

    ; Finalisation tls13-04 (Q2, I1)
    mov rcx, rsp
    add rcx, 0x40     ; Pointeur vers le contexte SHA
    mov rdx, rsp
    add rdx, 0xB0
    call 0xc300       ; tls13-04 écrit le condensé dans [rcx]

    ; Copie du résultat dans la sortie (Q1)
    mov rsi, rsp
    add rsi, 0xB0     ; condensé généré
    mov rdi, r13      ; destination
    mov rcx, 32       ; 32o
    rep movsb

    ; Succès (Q4)
    xor rax, rax      ; rax = 0
    jmp .fin

.erreur_session:
    mov rax, -4       ; E1
    jmp .fin

.erreur_hors_bornes:
    mov rax, -3       ; E2
    jmp .fin

.erreur_entree:
    mov rax, -5       ; E3
    jmp .fin

.erreur_systeme:
    ; E4 : GetLastError, sortie intacte (I4, I5)
    call qword ptr [rip + IAT_GetLastError]
    mov rax, -6       ; E4
    ; Le condensé n'est pas écrit (sortie intacte, Q1 préservé)
    jmp .fin

.vide:
    ; Cas vide : exécuter l'initialisation, puis finaliser directement (Q3)
    mov rcx, rsp
    add rcx, 0x40
    call 0xc000       ; tls13-01
    mov rcx, rsp
    add rcx, 0x40
    mov rdx, rsp
    add rdx, 0xB0
    call 0xc300       ; tls13-04
    ; Copie du résultat dans la sortie (Q1)
    mov rsi, rsp
    add rsi, 0xB0     ; condensé généré
    mov rdi, r13      ; destination
    mov rcx, 32       ; 32o
    rep movsb
    xor rax, rax      ; Q4
    jmp .fin

.fin:
    ; Restauration des registres non volatils (D7/D35)
    add rsp, 0xe0
    pop rsi
    pop rdi
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    pop rbp
    ret               ; NOP implicite (0x90, I5)
    ret               ; NOP implicite (0x90, I5)

; Données
section .data
NULL equ 0
```
