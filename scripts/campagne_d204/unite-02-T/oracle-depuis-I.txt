LE MODÈLE (la sémantique du contrat, une fonction) :

  def modele_lire(fichier, decalage, journal):
      # journal : la liste des entrées loupe-16 (décalage u64, octet u8)
      taille = len(fichier)
      if decalage % 16 != 0:  return (-5, None, None)   # E2 : la grille
      if decalage >= taille:  return (-3, None, None)   # E3 : hors-bornes
      remplissage = min(384, taille - decalage)         # Q1 : la double borne
      tampon = bytearray(fichier[decalage:decalage + remplissage])
      for dec, octet in journal:                        # Q2 : le recouvrement
          if decalage <= dec < decalage + remplissage:
              tampon[dec - decalage] = octet
      return (0, bytes(tampon), remplissage)            # Q3 : rax = 0

  ET LA SESSION D'ABORD (P1) : sans session ouverte, verdict -4, RIEN lu —
  le modèle ne court même pas (E1 précède tout).
