================================================================================
D20 (la ronde 2 du MD5) : « f = G(c, d, b); » — G est ta macro MAJORITÉ ;
  la ronde 2 du MD5 est la SÉLECTION : f = (d & b) | (~d & c).
D21 (la ronde 4 du MD5) : « f = Ch(b, c, d); » — Ch est (b&c)^(~b&d) ;
  la ronde 4 du MD5 est : f = c ^ (b | ~d).
D22 (FATAL, la rotation) : « ROTL32(a + f + T[i] + w[g], S[i]) » — ta
  table S ne porte que 16 valeurs : S[i] vaut ZÉRO pour toute ronde

  contrat Q1 : GENERIC_READ | GENERIC_WRITE (Loupe est seul maître des

void md5_transform(uint32_t state[4], const uint8_t data[64]) {
    uint32_t a = state[0], b = state[1], c = state[2], d = state[3];
    uint32_t w[16];

    for (int i = 0; i < 16; ++i) {
        w[i] = (uint32_t)data[i * 4] | (uint32_t)data[i * 4 + 1] << 8 |
               (uint32_t)data[i * 4 + 2] << 16 | (uint32_t)data[i * 4 + 3] << 24;
    }

    for (int i = 0; i < 64; ++i) {
        uint32_t f, g;
        if (i < 16) { f = F(b, c, d); g = i; }
        else if (i < 32) { f = G(c, d, b); g = (5 * i + 1) % 16; }
        else if (i < 48) { f = H(b, c, d); g = (3 * i + 5) % 16; }
        else { f = Ch(b, c, d); g = (7 * i) % 16; }

        uint32_t temp = d;
        d = c;
        c = b;
        b = b + ROTL32(a + f + T[i] + w[g], S[i]);
        a = temp;
    }

    state[0] += a;
    state[1] += b;
    state[2] += c;
    state[3] += d;
}

void sha256_transform(uint32_t state[8], const uint8_t data[64]) {
    uint32_t a = state[0], b = state[1], c = state[2], d = state[3];
    uint32_t e = state[4], f = state[5], g = state[6], h = state[7];
    uint32_t w[64];

    for (int i = 0; i < 16; ++i) {
        w[i] = (uint32_t)data[i * 4] << 24 | (uint32_t)data[i * 4 + 1] << 16 |
               (uint32_t)data[i * 4 + 2] << 8 | (uint32_t)data[i * 4 + 3];
    }
    for (int i = 16; i < 64; ++i) {
        uint32_t s0 = ROTR32(w[i - 15], 7) ^ ROTR32(w[i - 15], 18) ^ (w[i - 15] >> 3);
        uint32_t s1 = ROTR32(w[i - 2], 17) ^ ROTR32(w[i - 2], 19) ^ (w[i - 2] >> 10);
        w[i] = w[i - 16] + s0 + w[i - 7] + s1;
    }

    for (int i = 0; i < 64; ++i) {
        uint32_t S1 = ROTR32(e, 6) ^ ROTR32(e, 11) ^ ROTR32(e, 25);
        uint32_t ch = Ch(e, f, g);
        uint32_t temp1 = h + S1 + ch + K[i] + w[i];
        uint32_t S0 = ROTR32(a, 2) ^ ROTR32(a, 13) ^ ROTR32(a, 22);
        uint32_t maj = ((a & b) | (a & c)) | (b & c);
        uint32_t temp2 = S0 + maj;

        h = g;
        g = f;
        f = e;
        e = d + temp1;
        d = c;
        c = b;
        b = a;
        a = temp1 + temp2;
    }

    state[0] += a; state[1] += b; state[2] += c; state[3] += d;
    state[4] += e; state[5] += f; state[6] += g; state[7] += h;
}

void md5_absorbe(ctx_md5 *c, const uint8_t *d, uint32_t n) {
    for (uint32_t k = 0; k < n; ++k) {
        c->tampon[c->rempli++] = d[k];
        c->total++;
        if (c->rempli == 64) {
            md5_transform(c->etat, c->tampon);
            c->rempli = 0;
        }
    }
}

void sha256_absorbe(ctx_sha256 *c, const uint8_t *d, uint32_t n) {
    for (uint32_t k = 0; k < n; ++k) {
        c->tampon[c->rempli++] = d[k];
        c->total++;
        if (c->rempli == 64) {
            sha256_transform(c->etat, c->tampon);
            c->rempli = 0;
        }
    }
}

void md5_final(ctx_md5 *c, uint8_t sortie[16]) {
    uint64_t bits = c->total * 8;
    c->tampon[c->rempli++] = 0x80;
    if (c->rempli > 56) {
        for (int i = c->rempli; i < 64; ++i) c->tampon[i] = 0;
        md5_transform(c->etat, c->tampon);
        c->rempli = 0;
    }
    for (int i = c->rempli; i < 56; ++i) c->tampon[i] = 0;
    for (int i = 0; i < 8; ++i) {
        c->tampon[56 + i] = (uint8_t)(bits >> (i * 8));
    }
    md5_transform(c->etat, c->tampon);
    for (int i = 0; i < 4; ++i) {
        uint32_t s = c->etat[i];
        sortie[i * 4 + 0] = (uint8_t)(s);
        sortie[i * 4 + 1] = (uint8_t)(s >> 8);
        sortie[i * 4 + 2] = (uint8_t)(s >> 16);
        sortie[i * 4 + 3] = (uint8_t)(s >> 24);
    }
}

void sha256_final(ctx_sha256 *c, uint8_t sortie[32]) {
    uint64_t bits = c->total * 8;
    c->tampon[c->rempli++] = 0x80;
    if (c->rempli > 56) {
        for (int i = c->rempli; i < 64; ++i) c->tampon[i] = 0;
        sha256_transform(c->etat, c->tampon);
        c->rempli = 0;
    }
    for (int i = c->rempli; i < 56; ++i) c->tampon[i] = 0;
    for (int i = 0; i < 8; ++i) {
        c->tampon[56 + i] = (uint8_t)(bits >> ((7 - i) * 8));

        uint32_t s = c->etat[i];

        sortie[i * 4 + 2] = (uint8_t)(s >> 8);

}
